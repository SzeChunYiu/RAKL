% Standalone RAKL recursive research-loop diagram (Paper I).
% Build:  pdflatex recursive_loop.tex  -> recursive_loop.pdf
\documentclass[border=6pt]{standalone}
\usepackage{tikz}
\usepackage{helvet}\renewcommand{\familydefault}{\sfdefault}
\usetikzlibrary{positioning,arrows.meta,shapes.geometric,calc}
\begin{document}
\begin{tikzpicture}[
  font=\small,
  >={Stealth[length=2.4mm]},
  proc/.style={rounded corners=2pt, draw=black!55, line width=0.5pt, align=center,
               inner sep=4pt, minimum height=9mm, minimum width=20mm, fill=white},
  dec/.style={diamond, aspect=2.1, draw=black!60, line width=0.5pt, align=center,
              inner sep=1pt, fill=blue!5},
  term/.style={rounded corners=6pt, draw=black!70, line width=0.7pt, align=center,
               inner sep=4pt, minimum height=9mm, fill=blue!6},
  hist/.style={proc, fill=black!5},
  flow/.style={->, line width=0.6pt, draw=black!65},
  lbl/.style={font=\scriptsize\itshape, text=black!60},
  node distance=8mm and 12mm,
]
% main pipeline
\node[proc] (prop) {propose\\{\scriptsize $G_\theta$ / OWMD}};
\node[proc, right=of prop] (gate) {workspace\\gate};
\node[proc, right=of gate] (ver) {verify\\{\scriptsize typed outcome}};
\node[dec, right=14mm of ver] (vq) {verified?};
\node[proc, right=14mm of vq] (prom) {promote /\\archive};
\node[proc, right=of prom] (rec) {record\\growth $\Delta$};
\node[dec, right=14mm of rec] (sq) {$\Delta$ flat?};
\node[term, right=14mm of sq] (rel) {release\\{\scriptsize (bounded-saturated)}};

% refuted branch -> negative history (below vq), rejoining at record
\node[hist, below=12mm of vq] (hist) {negative\\history $\mathcal H^-$};

% edges
\draw[flow] (prop) -- (gate);
\draw[flow] (gate) -- (ver);
\draw[flow] (ver) -- (vq);
\draw[flow] (vq) -- node[lbl,above]{certificate} (prom);
\draw[flow] (vq) -- node[lbl,right]{refuted} (hist);
\draw[flow] (prom) -- (rec);
\draw[flow] (hist) -| (rec);
\draw[flow] (rec) -- (sq);
\draw[flow] (sq) -- node[lbl,above]{yes} (rel);

% reopen loop back to propose (along the bottom)
\draw[flow, draw=black!55]
   (sq.south) |- ($(hist.south)-(0,8mm)$) -| node[lbl,below,pos=0.25]{no: reopen fibers} (prop.south);
\end{tikzpicture}
\end{document}
